# Web framework: defines routes, handles requests/responses, generates the /docs Swagger UI
fastapi

# ASGI server that actually runs the FastAPI app (what "uvicorn app.main:app --reload" uses)
uvicorn

# ORM: lets us define Python classes (like Notice) that map to PostgreSQL tables,
# and write queries as Python instead of raw SQL
sqlalchemy

# PostgreSQL driver — the actual library SQLAlchemy uses under the hood to talk to Postgres
psycopg2-binary

# Loads variables from the .env file (like DATABASE_URL) into the environment at runtime
python-dotenv

# Used by FastAPI to define and validate request bodies (e.g. NoticeCreateRequest)
pydantic
